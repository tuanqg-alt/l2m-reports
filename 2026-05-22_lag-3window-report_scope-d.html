(async () => {
  const DATE = '2026-05-23';  // <- THAY DATE NAY
  const fname = `lag_raw_${DATE}.txt`;
  if (!window._lag_armed) {
    const d = Object.getOwnPropertyDescriptor(HTMLTextAreaElement.prototype, 'value');
    Object.defineProperty(HTMLTextAreaElement.prototype, 'value', {
      set(v) { if (v && v.length > 100) window._obs_data = v; d.set.call(this, v); },
      get() { return d.get.call(this); }
    });
    window._lag_armed = true;
  }
  window._obs_data = '';
  if (!window.LogSearch?.doCopyAll) { console.error('LogSearch not found'); return; }
  LogSearch.doCopyAll();
  const t = Date.now();
  while (!window._obs_data && Date.now()-t < 12000) await new Promise(r=>setTimeout(r,150));
  if (!window._obs_data) { console.error('Timeout'); return; }
  const blob = new Blob([window._obs_data], {type:'text/plain;charset=utf-8'});
  const a = document.createElement('a'); a.href = URL.createObjectURL(blob);
  a.download = fname; a.click();
  console.log(`Downloaded ${fname}`);
})();
